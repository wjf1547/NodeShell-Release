## 14.1 示例一：待办清单（Todo）

覆盖：state → render 模式、受控输入、`$prefs` 持久化、`list` + `footer`、返回键拦截。

```js
// apps/todo/main.js
'use strict';

const state = {
  input: '',
  items: JSON.parse(pref('prefs.items', '[]')),   // [{id, text, done}]
  filter: 'all',                                  // all | active | done
};

function pref(k, d) { const v = $prefs.get(k); return (v === undefined || v === null || v === '') ? d : v; }
function save() { $prefs.set('prefs.items', JSON.stringify(state.items)); }
function genId() { return Date.now().toString(36) + Math.random().toString(36).slice(2, 6); }

function visible() {
  if (state.filter === 'active') return state.items.filter(t => !t.done);
  if (state.filter === 'done')   return state.items.filter(t => t.done);
  return state.items;
}

function render() {
  const list = visible();
  const left = state.items.filter(t => !t.done).length;

  $ui.render({
    type: 'root',
    props: { title: '待办 (' + left + ')', id: 'page-todo' },
    children: [
      // ---- 输入栏 ----
      { type:'card', children:[
        { type:'row', props:{ scroll:false, gap:8 }, children:[
          { type:'input', id:'newTodo', props:{
              value: state.input, placeholder:'要做什么?',
              onChange:'in', onSubmit:'add' } },
          { type:'button', props:{ text:'添加', onClick:'add' } },
        ]},
      ]},

      // ---- 筛选 ----
      { type:'row', props:{ gap:8 }, children:[
        { type:'chip', id:'f-all',    props:{ text:'全部 ' + state.items.length, selected: state.filter==='all',    onClick:'filter' } },
        { type:'chip', id:'f-active', props:{ text:'未完成 ' + left,             selected: state.filter==='active', onClick:'filter' } },
        { type:'chip', id:'f-done',   props:{ text:'完成 ' + (state.items.length-left), selected: state.filter==='done', onClick:'filter' } },
      ]},

      // ---- 列表 ----
      list.length
        ? { type:'list', props:{
              items: list.map(t => ({
                id: t.id,
                icon: t.done ? '✅' : '⬜',
                title: t.text,
                subtitle: t.done ? '已完成' : '',
                onClick: 'toggle',
                more: [ { text:'删除', onClick:'del' } ],
              })),
              footer: [
                { type:'space', props:{ height:6 } },
                { type:'actions', props:{ align:'fill', items:[
                  { text:'清除已完成', variant:'text', onClick:'clearDone' },
                ]}},
              ],
          }}
        : { type:'empty', props:{ icon:'🎉', text: state.filter==='all' ? '还没有待办，加一个吧' : '这个分类是空的' } },
    ],
  });
}

// ---------- 事件 ----------
$ui.on('in', (e) => { state.input = e.value; render(); });

$ui.on('add', () => {
  const text = String(state.input || '').trim();
  if (!text) { $app.toast('先输入内容'); return; }
  state.items.unshift({ id: genId(), text, done: false });
  state.input = '';
  save(); render();
});

$ui.on('toggle', (e) => {
  const t = state.items.find(x => x.id === e.id);
  if (t) { t.done = !t.done; save(); render(); }
});

$ui.on('del', (e) => {
  const i = state.items.findIndex(x => x.id === e.id);   // ⚠️ 按索引删
  if (i >= 0) { state.items.splice(i, 1); save(); render(); $app.toast('已删除'); }
});

$ui.on('filter', (e) => {
  state.filter = e.selected ? e.id.replace('f-', '') : 'all';
  render();
});

$ui.on('clearDone', async () => {
  const ok = await $ui.confirm({ title:'确认', text:'清除所有已完成的待办?' });
  if (ok) {
    state.items = state.items.filter(t => !t.done);
    save(); render(); $app.toast('已清除');
  }
});

// ---------- 启动 ----------
$app.on('launch', render);
```

**这个例子用到的模式，90% 的应用都要用**：state / render / `e.id` 回查 / `prefs.` 落盘 / 按索引删。

## 14.2 示例二：视频浏览器

覆盖：列表 + 搜索、`$http` 请求 + 超时、翻页判 `hasMore`、`$media.open`、带 headers 播放。

```js
// apps/vlist/main.js
'use strict';

const API = 'https://api.example.com';
const PAGE = 20;

const state = {
  kw: '',
  items: [],          // [{id, title, cover, url, raw, duration, site}]
  loading: false,
  loaded: false,
  hasMore: true,
  page: 0,
};

function withTimeout(p, ms) {
  return Promise.race([p, new Promise((_, rj) => setTimeout(() => rj(new Error('请求超时')), ms))]);
}

function render() {
  $ui.render({
    type:'root',
    props:{ title: '视频库', id:'page-list' },
    children:[
      { type:'search', id:'kw', props:{
          value: state.kw, placeholder:'搜索视频',
          onChange:'kw', onSubmit:'search' } },

      state.loading && state.items.length === 0
        ? { type:'loading', props:{ text:'加载中…' } }
        : state.items.length
          ? { type:'list', id:'mainList', props:{
                items: state.items.map(v => ({
                  id: v.id,
                  title: v.title,
                  subtitle: v.site + ' · ' + v.duration,
                  avatar: v.cover,
                  badge: 'HD',
                  right: '▶',
                  onClick: 'play',
                })),
                refreshing: state.loading,
                onRefresh: 'refresh',
                onEndReached: 'more',
                footer: [ { type:'space', props:{ height:16 } } ],
            }}
          : { type:'empty', props:{ icon:'🔍', text: state.kw ? '没有找到「' + state.kw + '」' : '搜点什么看看' } },
    ],
  });
}

async function load(fresh) {
  if (!fresh && state.loaded && !state.hasMore) return;    // ⚠️ 必须判
  state.loading = true; render();
  try {
    const p = fresh ? 0 : state.page + 1;
    const r = await withTimeout($http.get(
      API + '/search?kw=' + encodeURIComponent(state.kw) + '&page=' + p + '&size=' + PAGE,
      { headers:{ 'User-Agent':'Mozilla/5.0' }, timeout: 15000 }
    ), 15000);
    const list = (r.json && r.json.list) || [];
    state.hasMore = list.length === PAGE;
    state.loaded  = true;
    state.page    = fresh ? 1 : state.page + 1;
    state.items   = fresh ? list : state.items.concat(list);
    if (fresh && list.length === 0) $app.toast('没有结果');
  } catch (e) {
    $app.toast('加载失败: ' + e.message);
  }
  state.loading = false;
  render();
}

$ui.on('kw',     (e) => { state.kw = e.value; });          // 输入只更新 state
$ui.on('search', (e) => { state.kw = e.value; load(true); });
$ui.on('refresh',() => load(true));
$ui.on('more',   () => load(false));

$ui.on('play', (e) => {
  const v = state.items.find(x => x.id === e.id);          // ⚠️ e.id 回查
  if (!v) return;
  // ⚠️ 不要 await —— 播放期间 Promise 一直挂起
  $media.open({
    url: v.url,
    infoUrl: v.raw,            // ℹ 弹窗显示原始链接（解密/转换前）
    type: 'video',
    title: v.title,
    poster: v.cover,
    autoPlay: true,
  }).then(r => {
    console.log('播放关闭, finished =', r.finished);
  }).catch(err => {
    console.error('播放失败:', err.message);
  });
});

$app.on('launch', () => load(true));
```

**关键点回顾**：`!fresh && loaded && !hasMore` 早退 / `$media.open` 不 await / 请求带超时。

## 14.3 示例三：密码管理器

覆盖：`input type:password`、`list` + `more` 菜单、`$ui.prompt` 对话框、`$prefs` 存敏感数据、按索引删。

```js
// apps/pwdbox/main.js
'use strict';

const KEY = 'prefs.entries';

const state = {
  entries: load(),        // [{id, name, user, pwd, note}]
  unlocked: false,
};

function load() {
  try {
    const r = JSON.parse($prefs.get(KEY, '[]'));
    return Array.isArray(r) ? r : [];
  } catch (e) { return []; }
}
function save() { $prefs.set(KEY, JSON.stringify(state.entries)); }
function genId() { return Date.now().toString(36) + Math.random().toString(36).slice(2,6); }

function render() {
  if (!state.unlocked) { renderLock(); return; }

  $ui.render({
    type:'root',
    props:{ title:'密码箱 (' + state.entries.length + ')', id:'page-list' },
    children:[
      state.entries.length
        ? { type:'list', props:{
              items: state.entries.map(x => ({
                id: x.id,
                icon: '🔑',
                title: x.name,
                subtitle: x.user,
                right: '👁',
                onClick: 'view',
                more: [
                  { text:'复制密码', onClick:'copy' },
                  { text:'编辑',     onClick:'edit' },
                  { text:'删除',     onClick:'del'  },
                ],
              })),
          }}
        : { type:'empty', props:{ icon:'🔐', text:'还没有条目，点右下角添加' } },
      { type:'fab', props:{ text:'＋', onClick:'add' } },
      { type:'space', props:{ height:72 } },
    ],
  });
}

function renderLock() {
  $ui.render({
    type:'root', props:{ title:'密码箱' },
    children:[
      { type:'card', children:[
        { type:'text', props:{ text:'输入主密码解锁', size:14, bold:true } },
        { type:'space', props:{ height:8 } },
        { type:'input', id:'master', props:{
            type:'password', placeholder:'主密码',
            onChange:'m', onSubmit:'unlock' } },
        { type:'space', props:{ height:8 } },
        { type:'button', props:{ text:'解锁', onClick:'unlock' } },
      ]},
    ],
  });
}

// ---------- 事件 ----------
$ui.on('m',      (e) => { state._m = e.value; render(); });
$ui.on('unlock', () => {
  if (String(state._m || '') === '1234') { state.unlocked = true; state._m = ''; render(); }
  else $app.toast('主密码错误');
});

$ui.on('add', async () => {
  const name = await $ui.prompt({ title:'新增', text:'名称(如 GitHub)' });
  if (!name) return;
  const user = await $ui.prompt({ title:'新增', text:'账号' });
  if (user === null) return;
  const pwd  = await $ui.prompt({ title:'新增', text:'密码' });
  if (pwd === null) return;
  state.entries.unshift({ id: genId(), name, user, pwd: pwd || '', note:'' });
  save(); render(); $app.toast('已添加');
});

$ui.on('view', (e) => {
  const x = state.entries.find(v => v.id === e.id);
  if (x) $ui.alert({ title:x.name, text:'账号: ' + x.user + '\n密码: ' + x.pwd });
});

$ui.on('copy', (e) => {
  const x = state.entries.find(v => v.id === e.id);
  if (x) { $app.copy(x.pwd); $app.toast('密码已复制'); }
});

$ui.on('edit', async (e) => {
  const i = state.entries.findIndex(v => v.id === e.id);
  if (i < 0) return;
  const np = await $ui.prompt({ title:'修改密码', text: state.entries[i].name,
                                default: state.entries[i].pwd });
  if (np === null) return;
  state.entries[i].pwd = np;
  save(); render(); $app.toast('已保存');
});

$ui.on('del', async (e) => {
  const i = state.entries.findIndex(v => v.id === e.id);
  if (i < 0) return;
  const ok = await $ui.confirm({ title:'删除', text:'确定删除「' + state.entries[i].name + '」?' });
  if (!ok) return;
  state.entries.splice(i, 1);        // ⚠️ 按索引删
  save(); render(); $app.toast('已删除');
});

$app.on('launch', render);
```
